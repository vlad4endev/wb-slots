### ✅ 1. Структура try-catch (КРИТИЧНО)

**Файл:** `src/lib/services/auto-booking-service.ts`

**ДО:**
```typescript
try {
  const { page } = await initBrowser();
} // ← try закрывается

const result = await doSomething(page); // ❌ page undefined!
```

**ПОСЛЕ:**
```typescript
let stopPeriodicScreenshots: (() => void) | null = null;

try {
  const { page } = await initBrowser();
  
  // ✅ ВСЁ внутри try!
  stopPeriodicScreenshots = await monitor.startPeriodicScreenshots(page, 15000);
  const sessionResult = await validateAndRestoreSession(page, userId);
  const navResult = await enhancedNavigateToSupplies(page, timeoutConfig);
  // ... весь остальной код
  
} finally {
  if (stopPeriodicScreenshots) {
    stopPeriodicScreenshots();
  }
  await cleanup();
}
```

### ✅ 5. authStatus scope (КРИТИЧНО)

**Файл:** `src/lib/services/auto-booking-service.ts`, строки 860-923

**ДО:**
```typescript
if (hasAuthCookie) {
  const authStatus = await checkAuth(page);
}

return {
  authenticated: authStatus.isAuthenticated // ❌ authStatus not defined!
};
```

**ПОСЛЕ:**
```typescript
let authCheckResult = {
  isAuthenticated: false,
  userInfo: null,
  checked: false
};

if (hasAuthCookie) {
  const authStatus = await checkAuth(page);
  authCheckResult.isAuthenticated = authStatus.isAuthenticated;
  authCheckResult.userInfo = authStatus.userInfo;
  authCheckResult.checked = true;
}

return {
  authenticated: authCheckResult.isAuthenticated // ✅ Работает!
};
```
